//////////////////////////////////////////////////////

Enum user_status {
  PENDING
  ACTIVE
  LOCKED
  DISABLED
}


Enum tenant_status {
  ACTIVE
  SUSPENDED
  DELETED
}


Enum workspace_status {
  ACTIVE
  ARCHIVED
  SUSPENDED
}


Enum project_status {
  DRAFT
  ACTIVE
  COMPLETED
  ARCHIVED
}


Enum task_status {
  BACKLOG
  TODO
  IN_PROGRESS
  BLOCKED
  REVIEW
  DONE
  CANCELLED
}


Enum priority {
  LOW
  MEDIUM
  HIGH
  URGENT
}


Enum event_status {
  PENDING
  PROCESSING
  COMPLETED
  FAILED
}



//////////////////////////////////////////////////////
// TENANT
//////////////////////////////////////////////////////

Table tenants {

  id uuid [pk]

  name varchar(255)

  slug varchar(255) [unique]

  status tenant_status


  version int [default:1]


  created_at timestamp

  updated_at timestamp


  deleted_at timestamp

}



//////////////////////////////////////////////////////
// IDENTITY
//////////////////////////////////////////////////////

Table users {

  id uuid [pk]


  tenant_id uuid [not null]


  email varchar(255) [unique]

  password_hash varchar(255)


  first_name varchar(100)

  last_name varchar(100)

  avatar_url text


  status user_status


  email_verified boolean


  last_login_at timestamp


  version int [default:1]


  created_at timestamp

  updated_at timestamp


  created_by uuid

  updated_by uuid


  deleted_at timestamp


  indexes {

    tenant_id

    email

  }

}



Table sessions {

  id uuid [pk]


  user_id uuid


  refresh_token text


  expires_at timestamp


  revoked boolean


  created_at timestamp

}



Table oauth_accounts {

  id uuid [pk]


  user_id uuid


  provider varchar(50)


  provider_account_id varchar(255)


  created_at timestamp

}
